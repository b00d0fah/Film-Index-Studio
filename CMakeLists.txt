cmake_minimum_required(VERSION 3.24)
project(FilmIndexStudio VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent Test)
file(GLOB_RECURSE FILM_ASSETS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "assets/film_strips/135/*.png")
list(APPEND FILM_ASSETS assets/logo.png assets/translations/qtbase_zh_CN.qm)
add_library(index_core STATIC src/document.cpp src/renderer.cpp)
target_include_directories(index_core PUBLIC src)
target_link_libraries(index_core PUBLIC Qt6::Widgets)
add_executable(FilmIndexStudio WIN32 src/main.cpp src/window.cpp)
if(WIN32)
  target_sources(FilmIndexStudio PRIVATE assets/app.rc)
endif()
qt_add_resources(FilmIndexStudio templates PREFIX "/" FILES ${FILM_ASSETS})
target_link_libraries(FilmIndexStudio PRIVATE index_core Qt6::Concurrent)
if(MINGW)
  target_compile_options(index_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(FilmIndexStudio PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(index_tests tests/test_core.cpp)
qt_add_resources(index_tests templates_test PREFIX "/" FILES ${FILM_ASSETS})
target_link_libraries(index_tests PRIVATE index_core Qt6::Test)
add_test(NAME index_core_tests COMMAND index_tests -platform offscreen)
add_executable(window_tests tests/test_window.cpp src/window.cpp)
qt_add_resources(window_tests templates_window PREFIX "/" FILES ${FILM_ASSETS})
target_link_libraries(window_tests PRIVATE index_core Qt6::Concurrent Qt6::Test)
add_test(NAME index_window_tests COMMAND window_tests -platform offscreen)
